# Libo Insights — Cron Jobs Configuration
# Install: crontab cron_setup.txt
# View: crontab -l

# Resolve repo root at install time (edit path if deploying elsewhere)
LIBO_INSIGHTS_ROOT=/workspace

# Realtime — forex + energy every 30 minutes
*/30 * * * * $LIBO_INSIGHTS_ROOT/cron_realtime.sh

# Weekly — DWS dam levels every Monday 06:00 UTC
0 6 * * 1 $LIBO_INSIGHTS_ROOT/cron_weekly.sh

# Monthly — SARB repo + Stats SA CPI + property on 1st of month 05:00 UTC
0 5 1 * * $LIBO_INSIGHTS_ROOT/cron_monthly.sh

# Quarterly — full refresh all 10 topics (Jan, Apr, Jul, Oct) 04:00 UTC
0 4 1 1,4,7,10 * $LIBO_INSIGHTS_ROOT/cron_quarterly.sh

# Log cleanup — delete logs older than 30 days (daily 02:00 UTC)
0 2 * * * find $LIBO_INSIGHTS_ROOT/logs -name "*.log" -mtime +30 -delete
